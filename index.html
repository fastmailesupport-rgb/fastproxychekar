const apiKey = "3873a942a458600bebf58923b8163e04"; async function checkMbBalance() { const inputVal = document.getElementById('proxyInput').value.trim(); if(!inputVal) { alert("দয়া করে প্রক্সি দিন!"); return; } let rawUser = inputVal; if (inputVal.includes(':')) { const parts = inputVal.split(':'); rawUser = parts.length >= 3 ? parts[2] : parts[0]; } const cleanUser = rawUser.split('-')[0].trim(); document.getElementById('resultBox').style.display = 'block'; document.getElementById('remMb').innerText = 'Loading...'; const targetUrl = `https://9proxy.com/api/v1/sub-user/list?key=${apiKey}`; const proxyUrl = `https://corsproxy.io/?` + encodeURIComponent(targetUrl); try { let response = await fetch(proxyUrl); let data = await response.json(); if (data && data.data) { const matchedUser = data.data.find(u => u.username === cleanUser || cleanUser.includes(u.username)); if (matchedUser) { document.getElementById('remMb').innerText = `${matchedUser.gb_used || 0} MB used`; document.getElementById('totalLeft').innerText = `${matchedUser.gb_used || 0} MB`; } else { document.getElementById('remMb').innerText = `User not found`; } } } catch (error) { document.getElementById('remMb').innerText = 'API Fetch Failed (Block)'; } }
